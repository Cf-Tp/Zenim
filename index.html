API Documentation

Integrate Zenim Keys validation into any application

🔗 Endpoint

POST https://cf-tp.github.io/Zenim/apiv1/validate

🟡 JavaScript/Node.js

const ZenimAPI = {
    validate: async (key) => {
        const res = await fetch(
            'https://cf-tp.github.io/Zenim/apiv1/validate',
            {
                method: 'POST',
                headers: {
                    'Authorization': 'Bearer YOUR_TOKEN',
                    'Content-Type': 'application/json'
                },
                body: JSON.stringify({ key })
            }
        );
        return res.json();
    }
};

🔵 Lua (Roblox)

local HttpService = game:GetService("HttpService")

local function validateKey(key)
    local res = HttpService:RequestAsync({
        Url = "https://cf-tp.github.io/Zenim/apiv1/validate",
        Method = "POST",
        Headers = {
            Authorization = "Bearer TOKEN",
            ["Content-Type"] = "application/json"
        },
        Body = HttpService:JSONEncode({key = key})
    })
    return HttpService:JSONDecode(res.Body)
end

🟣 C# (.NET)

using System.Net.Http;
using System.Text;
using Newtonsoft.Json;

public async Task<bool> ValidateKey(string key)
{
    using var client = new HttpClient();
    var payload = JsonConvert.SerializeObject(new { key });
    var content = new StringContent(payload, Encoding.UTF8, "application/json");
    client.DefaultRequestHeaders.Add("Authorization", "Bearer TOKEN");
    var res = await client.PostAsync("https://cf-tp.github.io/Zenim/apiv1/validate", content);
    return res.IsSuccessStatusCode;
}
